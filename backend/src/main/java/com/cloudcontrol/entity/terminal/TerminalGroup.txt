package com.cloudcontrol.entity.terminal;

import jakarta.persistence.*;
import org.hibernate.annotations.CreationTimestamp;
import org.hibernate.annotations.UpdateTimestamp;

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;

/**
 * 终端组实体类
 * 用于管理播控终端的分组，支持层级结构
 */
@Entity
@Table(name = "terminal_groups")
public class TerminalGroup {
    
    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    /**
     * 终端组名称
     */
    @Column(name = "name", nullable = false, length = 100)
    private String name;

    /**
     * 终端组描述
     */
    @Column(name = "description", length = 500)
    private String description;

    /**
     * 父级终端组 - 支持层级结构
     */
    @ManyToOne
    @JoinColumn(name = "parent_id")
    private TerminalGroup parent;
    
    /**
     * 子终端组列表
     */
    // 一对多，按照parent关联，级联删除，级联更新
    // 与parent字段关联
    @OneToMany(mappedBy = "parent", cascade = CascadeType.ALL, fetch = FetchType.LAZY)
    private List<TerminalGroup> children = new ArrayList<>();
    
    /**
     * 该组下的终端列表
     */
    @OneToMany(mappedBy = "terminalGroup", cascade = CascadeType.ALL, fetch = FetchType.LAZY)
    private List<Terminal> terminals = new ArrayList<>();
    
    /**
     * 创建时间
     */
    @CreationTimestamp
    @Column(name = "created_at", nullable = false, updatable = false)
    private LocalDateTime createdAt;
    
    /**
     * 更新时间
     */
    @UpdateTimestamp
    @Column(name = "updated_at")
    private LocalDateTime updatedAt;
    
    /**
     * 排序字段
     */
    @Column(name = "sort_order")
    private Integer sortOrder;
    
    /**
     * 是否启用
     */
    @Column(name = "enabled", nullable = false)
    private Boolean enabled = true;
    
    // 构造方法
    public TerminalGroup() {}
    
    public TerminalGroup(String name, String description) {
        this.name = name;
        this.description = description;
    }
    
    public TerminalGroup(String name, String description, TerminalGroup parent) {
        this.name = name;
        this.description = description;
        this.parent = parent;
    }
    
    // Getter和Setter方法
    public Long getId() {
        return id;
    }
    
    public void setId(Long id) {
        this.id = id;
    }
    
    public String getName() {
        return name;
    }
    
    public void setName(String name) {
        this.name = name;
    }
    
    public String getDescription() {
        return description;
    }
    
    public void setDescription(String description) {
        this.description = description;
    }
    
    public TerminalGroup getParent() {
        return parent;
    }
    
    public void setParent(TerminalGroup parent) {
        this.parent = parent;
    }
    
    public List<TerminalGroup> getChildren() {
        return children;
    }
    
    public void setChildren(List<TerminalGroup> children) {
        this.children = children;
    }
    
    public List<Terminal> getTerminals() {
        return terminals;
    }
    
    public void setTerminals(List<Terminal> terminals) {
        this.terminals = terminals;
    }
    
    public LocalDateTime getCreatedAt() {
        return createdAt;
    }
    
    public void setCreatedAt(LocalDateTime createdAt) {
        this.createdAt = createdAt;
    }
    
    public LocalDateTime getUpdatedAt() {
        return updatedAt;
    }
    
    public void setUpdatedAt(LocalDateTime updatedAt) {
        this.updatedAt = updatedAt;
    }
    
    public Integer getSortOrder() {
        return sortOrder;
    }
    
    public void setSortOrder(Integer sortOrder) {
        this.sortOrder = sortOrder;
    }
    
    public Boolean getEnabled() {
        return enabled;
    }
    
    public void setEnabled(Boolean enabled) {
        this.enabled = enabled;
    }
    
    /**
     * 添加子组
     */
    public void addChild(TerminalGroup child) {
        children.add(child);
        child.setParent(this);
    }
    
    /**
     * 移除子组
     */
    public void removeChild(TerminalGroup child) {
        children.remove(child);
        child.setParent(null);
    }
    
}
